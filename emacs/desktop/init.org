  (file-name-directory
   (directory-file-name

     (directory-file-name star-config-directory))))
  "Root of the dotfiles checkout.")

(add-to-list 'load-path (expand-file-name "autoload" star-config-directory))
(add-to-list 'load-path (expand-file-name "lisp" star-dotfiles-root))

(let ((lisp-root (expand-file-name "lisp" star-dotfiles-root)))
  (when (file-directory-p lisp-root)
    (dolist (directory (directory-files lisp-root t "^[^.]" t))
      (when (file-directory-p directory)
        (add-to-list 'load-path directory)))))

(defun star--warn (format-string &rest args)
  (display-warning
   'star-emacs
   (apply #'format format-string args)
   :warning))

(defun star-project-root ()
  "Return the current project root or signal a useful error."
  (or
   (when (fboundp 'projectile-project-root)
     (ignore-errors (projectile-project-root)))
   (when-let* ((project (project-current nil)))
     (project-root project))
   (user-error "Couldn't detect a project")))

(defun star-capture-todo-file ()
  "Return the central todo capture file."
  (expand-file-name star-capture-todo-file org-directory))

  "Return the central notes capture file."
  (expand-file-name star-capture-notes-file org-directory))

(defun star--capture-local-root (path)
  "Return PATH at the nearest project root."
  (let ((filename (file-name-nondirectory path)))
    (expand-file-name
     filename
     (or
      (locate-dominating-file
       (file-truename default-directory)
       filename)
      (star-project-root)))))

(defun star-capture-project-todo-file ()
  (star--capture-local-root star-capture-todo-file))

(defun star-capture-project-notes-file ()
  (star--capture-local-root star-capture-notes-file))

(defun star-capture-project-changelog-file ()
  (star--capture-local-root star-capture-changelog-file))

(defun star--capture-ensure-heading (headings &optional initial-level)
  "Find or create HEADINGS for a central Org capture target."
  (if (null headings)
      (widen)
    (let ((level (or initial-level 1)))
      (if
          (and
           (re-search-forward
            (format
             org-complex-heading-regexp-format
             (regexp-quote (car headings)))
            nil
            t)
           (= (org-current-level) level))
          (progn
            (beginning-of-line)
            (org-narrow-to-subtree))
        (goto-char (point-max))
        (unless (and (bolp) (eolp))
          (insert "\n"))
        (insert (make-string level ?*) " " (car headings) "\n")
        (beginning-of-line 0))
      (star--capture-ensure-heading (cdr headings) (1+ level)))))

(defun star--capture-central-file (file project)
  "Select FILE and PROJECT as an Org capture target."
  (let ((file (expand-file-name file org-directory)))
    (set-buffer (org-capture-target-buffer file))
    (org-capture-put-target-region-and-position)
    (widen)
    (goto-char (point-min))
    (star--capture-ensure-heading
     (append
      (org-capture-get :parents)
      (list project (org-capture-get :heading))))))

(defun star-config--literate-source-p (file)
  "Return non-nil when FILE is an Org source that tangles generated output."
  (with-temp-buffer
    (insert-file-contents file)
    (goto-char (point-min))
    (re-search-forward "\\_<:tangle\\_>" nil t)))

(defun star-config-source-files ()
  "Return every literate source that owns native desktop generated output."
  (sort
   (seq-filter
    #'star-config--literate-source-p
    (directory-files-recursively star-config-directory "\\.org\\'"))
   #'string-lessp))

(defun star-config-sync ()
  "Tangle all native desktop sources and reload personal configuration."
  (interactive)
  (require 'ob-tangle)
  (dolist (source (star-config-source-files))
    (org-babel-tangle-file source))
  (star-config-reload))

(defun star-config-reload ()
  "Reload the generated personal configuration."
  (interactive)
  (load-file (expand-file-name "personal.el" star-config-directory))
  (message "Native Emacs personal configuration reloaded"))

(defun star-set-file-template (matcher &rest properties)
  "Register a simple auto-insert template for MATCHER.

PROPERTIES accepts :trigger or :template for a file name, and :mode for a
major-mode symbol.  Unsupported template metadata is reported instead of being
silently discarded."
  (require 'autoinsert)
  (let* ((template
          (or
           (plist-get properties :template)
           (plist-get properties :trigger)))
         (mode (plist-get properties :mode))
         (condition
          (cond
           ((stringp matcher) matcher)
           ((symbolp matcher) matcher)
           ((consp matcher) (or (car matcher) matcher))
           (t matcher))))
    (if (null template)
        (star--warn
         "File template %S has no :template/:trigger and needs native review"
         matcher)
      (let ((template-file
             (expand-file-name
              (format "%s" template)
              (expand-file-name "templates" star-config-directory))))
        (add-to-list
         'auto-insert-alist
         (cons
          (if mode (cons condition mode) condition)
          template-file))
        (auto-insert-mode 1)))))

(defun star-enable-babel-languages (languages)
  "Enable available Org Babel LANGUAGES and report unavailable backends."
  (require 'ob)
  (let (available)
    (dolist (entry languages)
      (let* ((language (car entry))
             (enabled (cdr entry))
             (library (format "ob-%s" language)))
        (when enabled
          (if
              (or
               (eq language 'emacs-lisp)
               (locate-library library))
              (push entry available)
            (star--warn
             "Org Babel backend %s is unavailable; requested by native config"
             library)))))
    (org-babel-do-load-languages
     'org-babel-load-languages
     (nreverse available))))
